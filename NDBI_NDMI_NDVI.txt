// Center map on the area of interest
Map.centerObject(aoi, 10);

// Import required packages
var palettes = require('users/gena/packages:palettes');

// Define date range and cloud cover threshold
var startDate = '2025-01-01';
var endDate = '2025-12-31';
var cloudCoverThreshold = 50; // Images with 50% cloud cover or less will be selected

// Define quarter (1 = Jan-Mar, 2 = Apr-Jun, 3 = Jul-Sep, 4 = Oct-Dec)
var quarter = 1; // Change this based on your date range

// Visualization parameters
var trueColorViz = {
  bands: ['SR_B4', 'SR_B3', 'SR_B2'],
  min: 0.0,
  max: 0.15

var ndviViz = {
  min: -1,
  max: 1,
  palette: ['blue', 'white', 'green']

  min: -1,
  max: 1,

var ndmiViz = {
  min: -1,
  max: 1,
  palette: ['white', 'cyan', 'blue']
};

var lstViz = {
  min: 25.00,
  max: 40.00,
  palette: palettes.colorbrewer.RdBu[9].reverse()
};

var ndsiViz = {
 min: -1,
  max: 1,
  palette: ['blue', 'white', 'green']

  min: -1,

  min: -1,
  max: 1,

  min: 0,
  max: 200,
  palette: ['green', 'yellow', 'orange', 'red', 'darkred']

// Function to apply scaling factors
function applyScaleFactors(image) {
  var opticalBands = image.select('SR_B.*').multiply(0.0000275).add(-0.2);
  var thermalBands = image.select('ST_B.*').multiply(0.00341802).add(149.0);
  return image.addBands(opticalBands, null, true)
              .addBands(thermalBands, null, true);
}

// Function to mask clouds and cloud shadows
function cloudMask(image) {
  var cloudShadowBitmask = (1 << 3);
  var cloudBitmask = (1 << 5);
  var qa = image.select('QA_PIXEL');
  var mask = qa.bitwiseAnd(cloudShadowBitmask).eq(0)
                .and(qa.bitwiseAnd(cloudBitmask).eq(0));
  return image.updateMask(mask);
}

// Function to calculate LST (Land Surface Temperature) in Celsius
function calculateLST(image) {
  var lst = image.select('ST_B10').multiply(0.00341802).add(149.0).subtract(273.15);
  return lst.rename('LST');
}

// Import and preprocess Landsat 8 imagery
var imageCollection = ee.ImageCollection("LANDSAT/LC08/C02/T1_L2")
                      .filterBounds(aoi)
                      .filterDate(startDate, endDate)
                      .filter(ee.Filter.lt('CLOUD_COVER', cloudCoverThreshold))
                      .map(applyScaleFactors)
                      .map(cloudMask);

// Print out the image IDs and dates used in the collection
print('Image IDs:', imageCollection.aggregate_array('system:index'));
print('Image Dates:', imageCollection.aggregate_array('DATE_ACQUIRED'));

// Create a median composite image
var image = imageCollection.median().clip(aoi);

// Add the processed image to the map with the specified visualization
Map.addLayer(image, trueColorViz, 'True Color 432');

// ------------------- NDVI -------------------
var ndvi = image.normalizedDifference(['SR_B5', 'SR_B4']).rename('NDVI');
Map.addLayer(ndvi, ndviViz, 'NDVI');

// ------------------- NDBI -------------------
var ndbi = image.normalizedDifference(['SR_B6', 'SR_B5']).rename('NDBI');
Map.addLayer(ndbi, ndbiViz, 'NDBI');

// ------------------- NDMI -------------------
var ndmi = image.normalizedDifference(['SR_B5', 'SR_B6']).rename('NDMI');
Map.addLayer(ndmi, ndmiViz, 'NDMI');

// ------------------- NDSI -------------------
var ndsi = image.normalizedDifference(['SR_B6', 'SR_B5']).rename('NDSI');
Map.addLayer(ndsi, ndsiViz, 'NDSI');

// ------------------- SAVI -------------------
// Calculate SAVI (Soil Adjusted Vegetation Index)
var L = 0.5;
var nir = image.select('SR_B5');
var red = image.select('SR_B4');

var savi = nir.subtract(red)
  .divide(nir.add(red).add(L))
  .multiply(1 + L)
  .rename('SAVI');
Map.addLayer(savi, saviViz, 'SAVI');

// ------------------- NDWI -------------------
// Calculate NDWI (Normalized Difference Water Index) using Green and NIR
var ndwi = image.normalizedDifference(['SR_B3', 'SR_B5']).rename('NDWI');
Map.addLayer(ndwi, ndwiViz, 'NDWI');

// ------------------- LST -------------------
// Calculate Land Surface Temperature
var lst = calculateLST(image);
Map.addLayer(lst, lstViz, 'LST');

// ------------------- Extract Individual Bands -------------------
var blue = image.select('SR_B2').rename('Blue');
var green = image.select('SR_B3').rename('Green');
var red_band = image.select('SR_B4').rename('Red');
var infrared = image.select('SR_B5').rename('Infrared');
var swir = image.select('SR_B6').rename('SWIR');

// ------------------- PM10 Calculation -------------------
// Formula: PM10 = 1 + NDVI - NDSI - SAVI + NDWI - LST - Blue - Green + Red + Infrared + SWIR + Year + Quarter
// Note: The formula includes "+2025-1" which is interpreted as Year + Quarter

var year = 2025; // Year of Landsat data

// Calculate PM10
var pm10 = ee.Image(1)  // Start with 1
  .add(ndvi)            // + NDVI
  .subtract(ndsi)       // - NDSI
  .subtract(savi)       // - SAVI
  .add(ndwi)            // + NDWI
  .subtract(lst)        // - LST
  .subtract(blue)       // - Blue
  .subtract(green)      // - Green
  .add(red_band)        // + Red
  .add(infrared)        // + Infrared
  .add(swir)            // + SWIR
  .add(year)            // + Year (2025)
  .add(quarter)         // + Quarter (1, 2, 3, or 4)
  .rename('PM10');

// Add PM10 layer to map
Map.addLayer(pm10, pm10Viz, 'PM10');
